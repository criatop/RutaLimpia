# api-gateway - RutaLimpia

Puerta de entrada unica (Spring Cloud Gateway). Simula **Amazon API Gateway** en el entorno local:
los clientes llaman solo al puerto 8080 y el gateway enruta a cada microservicio por nombre, consultando a Eureka.

## Requisitos
- Java 17, Maven 3.9+
- `eureka-server` y los microservicios en ejecucion

## Clonar y ejecutar
```bash
git clone <URL_DEL_REPOSITORIO>
cd api-gateway
mvn clean install
mvn spring-boot:run        # http://localhost:8080
```
O con el JAR: `mvn clean package && java -jar target/api-gateway-1.0.0.jar`

## Rutas (src/main/resources/application.yml)
| Path | Servicio |
|---|---|
| /auth/**, /usuarios/** | ms-auth |
| /solicitudes/**, /tipos-residuo/** | ms-solicitudes |
| /camiones/**, /zonas/** | ms-flota |
| /rutas/** | ms-rutas |

Ejemplo: `POST http://localhost:8080/solicitudes` llega a `ms-solicitudes` sin conocer su puerto.

package com.rutalimpia.gateway.filter;

import java.util.Arrays;
import java.util.List;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.cloud.gateway.filter.GatewayFilterChain;
import org.springframework.cloud.gateway.filter.GlobalFilter;
import org.springframework.core.Ordered;
import org.springframework.http.HttpHeaders;
import org.springframework.http.HttpMethod;
import org.springframework.http.HttpStatus;
import org.springframework.http.server.reactive.ServerHttpRequest;
import org.springframework.stereotype.Component;
import org.springframework.util.AntPathMatcher;
import org.springframework.web.reactive.function.client.WebClient;
import org.springframework.web.server.ServerWebExchange;

import com.rutalimpia.gateway.dto.TokenInfo;

import reactor.core.publisher.Mono;

/**
 * Authorizer del API Gateway: valida el JWT contra ms-auth antes de enrutar.
 * Las rutas publicas (registro/login/validacion) no requieren token.
 * Al validar, propaga X-Usuario-Id y X-Rol a los microservicios.
 */
@Component
public class JwtAuthFilter implements GlobalFilter, Ordered {

    private static final Logger log = LoggerFactory.getLogger(JwtAuthFilter.class);
    private static final AntPathMatcher MATCHER = new AntPathMatcher();

    private final WebClient webClient;
    private final boolean enabled;
    private final List<String> publicPaths;

    public JwtAuthFilter(WebClient.Builder loadBalancedWebClientBuilder,
                         @Value("${rutalimpia.security.enabled:true}") boolean enabled,
                         @Value("${rutalimpia.security.public-paths:/auth/register,/auth/login,/auth/validate}") String publicPaths) {
        this.webClient = loadBalancedWebClientBuilder.clone().baseUrl("http://ms-auth").build();
        this.enabled = enabled;
        this.publicPaths = Arrays.stream(publicPaths.split(","))
                .map(String::trim)
                .filter(s -> !s.isEmpty())
                .toList();
    }

    @Override
    public Mono<Void> filter(ServerWebExchange exchange, GatewayFilterChain chain) {
        if (!enabled) {
            return chain.filter(exchange);
        }
        String path = exchange.getRequest().getPath().value();
        if (HttpMethod.OPTIONS.equals(exchange.getRequest().getMethod()) || esPublica(path)) {
            return chain.filter(exchange);
        }

        String authorization = exchange.getRequest().getHeaders().getFirst(HttpHeaders.AUTHORIZATION);
        if (authorization == null || !authorization.startsWith("Bearer ")) {
            return noAutorizado(exchange);
        }

        return webClient.get()
                .uri("/auth/validate")
                .header(HttpHeaders.AUTHORIZATION, authorization)
                .retrieve()
                .bodyToMono(TokenInfo.class)
                .flatMap(info -> info != null && info.valido()
                        ? chain.filter(conIdentidad(exchange, info))
                        : noAutorizado(exchange))
                .onErrorResume(e -> {
                    log.warn("[AUTH] Token invalido para {}: {}", path, e.getMessage());
                    return noAutorizado(exchange);
                });
    }

    private ServerWebExchange conIdentidad(ServerWebExchange exchange, TokenInfo info) {
        ServerHttpRequest request = exchange.getRequest().mutate()
                .header("X-Usuario-Id", info.usuarioId() == null ? "" : String.valueOf(info.usuarioId()))
                .header("X-Rol", info.rol() == null ? "" : info.rol())
                .build();
        return exchange.mutate().request(request).build();
    }

    private boolean esPublica(String path) {
        return publicPaths.stream().anyMatch(p -> MATCHER.match(p, path));
    }

    private Mono<Void> noAutorizado(ServerWebExchange exchange) {
        exchange.getResponse().setStatusCode(HttpStatus.UNAUTHORIZED);
        return exchange.getResponse().setComplete();
    }

    @Override
    public int getOrder() {
        return -1;
    }
}

package com.rutalimpia.auth.config;

import java.util.List;

import org.springframework.boot.CommandLineRunner;
import org.springframework.stereotype.Component;

import com.rutalimpia.auth.model.Rol;
import com.rutalimpia.auth.repository.RolRepository;

import lombok.RequiredArgsConstructor;

/** Crea los roles base al iniciar si no existen. */
@Component
@RequiredArgsConstructor
public class DataInitializer implements CommandLineRunner {

    private final RolRepository rolRepository;

    @Override
    public void run(String... args) {
        for (String nombre : List.of("VECINO", "CONDUCTOR")) {
            if (rolRepository.findByNombre(nombre).isEmpty()) {
                rolRepository.save(new Rol(nombre));
            }
        }
    }
}

import org.springframework.web.bind.annotation.GetMapping;

import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestHeader;

import com.rutalimpia.auth.dto.AuthResponse;
import com.rutalimpia.auth.dto.LoginRequest;
import com.rutalimpia.auth.dto.RegisterRequest;
import com.rutalimpia.auth.dto.TokenInfo;
import com.rutalimpia.auth.dto.UsuarioResponse;
import com.rutalimpia.auth.service.AuthService;

import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;

@RestController
@RequestMapping("/auth")
@RequiredArgsConstructor
public class AuthController {

    private final AuthService authService;

    @PostMapping("/register")
    public ResponseEntity<UsuarioResponse> registrar(@Valid @RequestBody RegisterRequest req) {
        return ResponseEntity.status(HttpStatus.CREATED).body(authService.registrar(req));
    }

    @PostMapping("/login")
    public AuthResponse login(@Valid @RequestBody LoginRequest req) {
        return authService.login(req);
    }

    /** Lo puede usar el API Gateway (authorizer) para validar el JWT y conocer el rol. */
    @GetMapping("/validate")
    public TokenInfo validar(@RequestHeader(value = "Authorization", required = false) String authorization) {
        return authService.validar(authorization);
    }
}

package com.rutalimpia.auth.exception;

import java.util.stream.Collectors;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.method.annotation.MethodArgumentTypeMismatchException;
import org.springframework.web.servlet.resource.NoResourceFoundException;

@RestControllerAdvice
public class GlobalExceptionHandler {

    private static final Logger log = LoggerFactory.getLogger(GlobalExceptionHandler.class);

    @ExceptionHandler(NotFoundException.class)
    public ResponseEntity<ApiError> notFound(NotFoundException ex) {
        return build(HttpStatus.NOT_FOUND, ex.getMessage());
    }

    @ExceptionHandler(ConflictException.class)
    public ResponseEntity<ApiError> conflict(ConflictException ex) {
        return build(HttpStatus.CONFLICT, ex.getMessage());
    }

    @ExceptionHandler(BadRequestException.class)
    public ResponseEntity<ApiError> badRequest(BadRequestException ex) {
        return build(HttpStatus.BAD_REQUEST, ex.getMessage());
    }

    @ExceptionHandler(UnauthorizedException.class)
    public ResponseEntity<ApiError> unauthorized(UnauthorizedException ex) {
        return build(HttpStatus.UNAUTHORIZED, ex.getMessage());
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiError> validation(MethodArgumentNotValidException ex) {
        String msg = ex.getBindingResult().getFieldErrors().stream()
                .map(f -> f.getField() + ": " + f.getDefaultMessage())
                .collect(Collectors.joining("; "));
        return build(HttpStatus.BAD_REQUEST, msg);
    }

    @ExceptionHandler({HttpMessageNotReadableException.class, MethodArgumentTypeMismatchException.class})
    public ResponseEntity<ApiError> malformed(Exception ex) {
        return build(HttpStatus.BAD_REQUEST, "Solicitud mal formada o parametro invalido");
    }

    @ExceptionHandler(DataIntegrityViolationException.class)
    public ResponseEntity<ApiError> integrity(DataIntegrityViolationException ex) {
        return build(HttpStatus.CONFLICT, "Violacion de integridad de datos (duplicado o referencia invalida)");
    }

    @ExceptionHandler(NoResourceFoundException.class)
    public ResponseEntity<ApiError> noResource(NoResourceFoundException ex) {
        return build(HttpStatus.NOT_FOUND, "Recurso no encontrado");
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<ApiError> generic(Exception ex) {
        log.error("Error interno no controlado", ex);
        return build(HttpStatus.INTERNAL_SERVER_ERROR, "Error interno del servidor");
    }

    private ResponseEntity<ApiError> build(HttpStatus status, String message) {
        return ResponseEntity.status(status).body(ApiError.of(status.value(), status.getReasonPhrase(), message));
    }
}

package com.rutalimpia.auth.service;

import java.util.List;

import org.springframework.security.crypto.password.PasswordEncoder;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.rutalimpia.auth.dto.AuthResponse;
import com.rutalimpia.auth.dto.LoginRequest;
import com.rutalimpia.auth.dto.RegisterRequest;
import com.rutalimpia.auth.dto.TokenInfo;
import com.rutalimpia.auth.dto.UsuarioResponse;
import com.rutalimpia.auth.dto.UsuarioUpdateRequest;
import com.rutalimpia.auth.exception.BadRequestException;
import com.rutalimpia.auth.exception.ConflictException;
import com.rutalimpia.auth.exception.NotFoundException;
import com.rutalimpia.auth.exception.UnauthorizedException;
import com.rutalimpia.auth.model.Rol;
import com.rutalimpia.auth.model.Usuario;
import com.rutalimpia.auth.repository.RolRepository;
import com.rutalimpia.auth.repository.UsuarioRepository;

import io.jsonwebtoken.Claims;
import lombok.RequiredArgsConstructor;

@Service
@RequiredArgsConstructor
@Transactional
public class AuthService {

    private final UsuarioRepository usuarioRepository;
    private final RolRepository rolRepository;
    private final PasswordEncoder passwordEncoder;
    private final JwtService jwtService;

    public UsuarioResponse registrar(RegisterRequest req) {
        if (usuarioRepository.existsByEmail(req.email())) {
            throw new ConflictException("Ya existe un usuario con el email " + req.email());
        }
        String nombreRol = req.rol().trim().toUpperCase();
        Rol rol = rolRepository.findByNombre(nombreRol)
                .orElseThrow(() -> new BadRequestException("Rol invalido: " + req.rol() + " (use VECINO o CONDUCTOR)"));
        if ("CONDUCTOR".equals(nombreRol) && req.camionId() == null) {
            throw new BadRequestException("Un CONDUCTOR debe tener camionId");
        }
        Usuario u = new Usuario();
        u.setNombre(req.nombre());
        u.setEmail(req.email());
        u.setPassword(passwordEncoder.encode(req.password()));
        u.setRol(rol);
        u.setCamionId("CONDUCTOR".equals(nombreRol) ? req.camionId() : null);
        return toResponse(usuarioRepository.save(u));
    }

    @Transactional(readOnly = true)
    public AuthResponse login(LoginRequest req) {
        Usuario u = usuarioRepository.findByEmail(req.email())
                .orElseThrow(() -> new UnauthorizedException("Credenciales invalidas"));
        if (!passwordEncoder.matches(req.password(), u.getPassword())) {
            throw new UnauthorizedException("Credenciales invalidas");
        }
        return new AuthResponse(jwtService.generar(u), "Bearer", u.getId(), u.getRol().getNombre(), u.getCamionId());
    }

    @Transactional(readOnly = true)
    public TokenInfo validar(String authorizationHeader) {
        if (authorizationHeader == null || !authorizationHeader.startsWith("Bearer ")) {
            throw new UnauthorizedException("Falta el header Authorization: Bearer <token>");
        }
        Claims c = jwtService.parsear(authorizationHeader.substring(7));
        Number usuarioId = c.get("usuarioId", Number.class);
        Number camionId = c.get("camionId", Number.class);
        return new TokenInfo(true,
                usuarioId == null ? null : usuarioId.longValue(),
                c.getSubject(),
                c.get("rol", String.class),
                camionId == null ? null : camionId.longValue());
    }

    @Transactional(readOnly = true)
    public List<UsuarioResponse> listar() {
        return usuarioRepository.findAll().stream().map(this::toResponse).toList();
    }

    @Transactional(readOnly = true)
    public UsuarioResponse obtener(Long id) {
        return toResponse(buscar(id));
    }

    public UsuarioResponse actualizar(Long id, UsuarioUpdateRequest req) {
        Usuario u = buscar(id);
        u.setNombre(req.nombre());
        if ("CONDUCTOR".equals(u.getRol().getNombre())) {
            u.setCamionId(req.camionId());
        }
        return toResponse(usuarioRepository.save(u));
    }

    public void eliminar(Long id) {
        usuarioRepository.delete(buscar(id));
    }

    private Usuario buscar(Long id) {
        return usuarioRepository.findById(id)
                .orElseThrow(() -> new NotFoundException("Usuario no encontrado: " + id));
    }

    private UsuarioResponse toResponse(Usuario u) {
        return new UsuarioResponse(u.getId(), u.getNombre(), u.getEmail(), u.getRol().getNombre(), u.getCamionId());
    }
}

package com.rutalimpia.auth.service;

import java.nio.charset.StandardCharsets;
import java.util.Date;

import javax.crypto.SecretKey;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;

import com.rutalimpia.auth.exception.UnauthorizedException;
import com.rutalimpia.auth.model.Usuario;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.JwtException;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.security.Keys;

@Service
public class JwtService {

    private final SecretKey key;
    private final long expirationMs;

    public JwtService(@Value("${jwt.secret}") String secret,
                      @Value("${jwt.expiration-ms}") long expirationMs) {
        this.key = Keys.hmacShaKeyFor(secret.getBytes(StandardCharsets.UTF_8));
        this.expirationMs = expirationMs;
    }

    public String generar(Usuario u) {
        Date ahora = new Date();
        return Jwts.builder()
                .subject(u.getEmail())
                .claim("usuarioId", u.getId())
                .claim("rol", u.getRol().getNombre())
                .claim("camionId", u.getCamionId())
                .issuedAt(ahora)
                .expiration(new Date(ahora.getTime() + expirationMs))
                .signWith(key)
                .compact();
    }

    public Claims parsear(String token) {
        try {
            return Jwts.parser().verifyWith(key).build().parseSignedClaims(token).getPayload();
        } catch (JwtException | IllegalArgumentException e) {
            throw new UnauthorizedException("Token invalido o expirado");
        }
    }
}

# ms-flota - RutaLimpia

Flota. Gestion de camiones, zonas y disponibilidad.

## Requisitos
- Java 17
- Maven 3.9+
- MySQL 8 corriendo en `localhost:3306` (la BD `rutalimpia_flota` se crea sola)

## Clonar y ejecutar
```bash
git clone <URL_DEL_REPOSITORIO>
cd ms-flota

# (opcional) credenciales de BD; por defecto root / root
export DB_USER=root
export DB_PASSWORD=root

mvn clean install        # compila, ejecuta tests y empaqueta
mvn spring-boot:run      # levanta el servicio en http://localhost:8083
```

O ejecutando el JAR generado:
```bash
mvn clean package
java -jar target/ms-flota-1.0.0.jar
```

MySQL rapido con Docker:
```bash
docker run -d --name mysql-rl -e MYSQL_ROOT_PASSWORD=root -p 3306:3306 mysql:8
```

## Documentacion
Swagger UI: http://localhost:8083/swagger-ui.html

## Endpoints
| Metodo | Ruta | Descripcion |
|---|---|---|
| POST | /camiones | Crear camion |
| GET | /camiones | Listar camiones |
| GET | /camiones/disponibles | Listar camiones disponibles |
| GET | /camiones/{id} | Obtener camion |
| PUT | /camiones/{id} | Actualizar camion |
| PUT | /camiones/{id}/disponibilidad | Cambiar disponibilidad |
| PUT | /camiones/{id}/asignar-parada | Reservar una parada (uso interno) |
| DELETE | /camiones/{id} | Eliminar camion |
| GET/POST | /zonas | Listar / crear zonas |

## Estructura
```
controller/  -> API REST
service/     -> logica de negocio
repository/  -> acceso a datos (Spring Data JPA)
model/       -> entidades JPA
dto/         -> objetos de entrada/salida
exception/   -> manejo global de errores
config/      -> configuracion y datos iniciales
```

## Service discovery (Eureka)
Este servicio se registra en Eureka como `ms-flota`. Levanta primero `eureka-server` (http://localhost:8761).
Para ejecutarlo sin Eureka: `export EUREKA_ENABLED=false`.
En produccion se accede a traves del `api-gateway` (http://localhost:8080).

    }

    /** Compensacion: libera la parada reservada si no se pudo crear en ms-rutas. */
    @PutMapping("/{id}/liberar-parada")
    public CamionResponse liberarParada(@PathVariable Long id) {
        return camionService.liberarParada(id);
    }

    @DeleteMapping("/{id}")
    public ResponseEntity<Void> eliminar(@PathVariable Long id) {
        camionService.eliminar(id);
        return ResponseEntity.noContent().build();
    }
}

package com.rutalimpia.flota.exception;

import java.util.stream.Collectors;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.method.annotation.MethodArgumentTypeMismatchException;
import org.springframework.web.servlet.resource.NoResourceFoundException;

@RestControllerAdvice
public class GlobalExceptionHandler {

    private static final Logger log = LoggerFactory.getLogger(GlobalExceptionHandler.class);

    @ExceptionHandler(NotFoundException.class)
    public ResponseEntity<ApiError> notFound(NotFoundException ex) {
        return build(HttpStatus.NOT_FOUND, ex.getMessage());
    }

    @ExceptionHandler(ConflictException.class)
    public ResponseEntity<ApiError> conflict(ConflictException ex) {
        return build(HttpStatus.CONFLICT, ex.getMessage());
    }

    @ExceptionHandler(BadRequestException.class)
    public ResponseEntity<ApiError> badRequest(BadRequestException ex) {
        return build(HttpStatus.BAD_REQUEST, ex.getMessage());
    }

    @ExceptionHandler(UnauthorizedException.class)
    public ResponseEntity<ApiError> unauthorized(UnauthorizedException ex) {
        return build(HttpStatus.UNAUTHORIZED, ex.getMessage());
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiError> validation(MethodArgumentNotValidException ex) {
        String msg = ex.getBindingResult().getFieldErrors().stream()
                .map(f -> f.getField() + ": " + f.getDefaultMessage())
                .collect(Collectors.joining("; "));
        return build(HttpStatus.BAD_REQUEST, msg);
    }

    @ExceptionHandler({HttpMessageNotReadableException.class, MethodArgumentTypeMismatchException.class})
    public ResponseEntity<ApiError> malformed(Exception ex) {
        return build(HttpStatus.BAD_REQUEST, "Solicitud mal formada o parametro invalido");
    }

    @ExceptionHandler(DataIntegrityViolationException.class)
    public ResponseEntity<ApiError> integrity(DataIntegrityViolationException ex) {
        return build(HttpStatus.CONFLICT, "Violacion de integridad de datos (duplicado o referencia invalida)");
    }

    @ExceptionHandler(NoResourceFoundException.class)
    public ResponseEntity<ApiError> noResource(NoResourceFoundException ex) {
        return build(HttpStatus.NOT_FOUND, "Recurso no encontrado");
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<ApiError> generic(Exception ex) {
        log.error("Error interno no controlado", ex);
        return build(HttpStatus.INTERNAL_SERVER_ERROR, "Error interno del servidor");
    }

    private ResponseEntity<ApiError> build(HttpStatus status, String message) {
        return ResponseEntity.status(status).body(ApiError.of(status.value(), status.getReasonPhrase(), message));
    }
}

package com.rutalimpia.flota.service;

import java.util.List;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.rutalimpia.flota.dto.CamionRequest;
import com.rutalimpia.flota.dto.CamionResponse;
import com.rutalimpia.flota.exception.BadRequestException;
import com.rutalimpia.flota.exception.ConflictException;
import com.rutalimpia.flota.exception.NotFoundException;
import com.rutalimpia.flota.model.Camion;
import com.rutalimpia.flota.model.Zona;
import com.rutalimpia.flota.repository.CamionRepository;
import com.rutalimpia.flota.repository.ZonaRepository;

import lombok.RequiredArgsConstructor;

@Service
@RequiredArgsConstructor
@Transactional
public class CamionService {

    private final CamionRepository camionRepository;
    private final ZonaRepository zonaRepository;

    public CamionResponse crear(CamionRequest req) {
        if (camionRepository.existsByPatente(req.patente())) {
            throw new ConflictException("Ya existe un camion con patente " + req.patente());
        }
        Camion c = new Camion();
        aplicar(c, req);
        return toResponse(camionRepository.save(c));
    }

    @Transactional(readOnly = true)
    public List<CamionResponse> listar() {
        return camionRepository.findAll().stream().map(this::toResponse).toList();
    }

    @Transactional(readOnly = true)
    public List<CamionResponse> listarDisponibles() {
        return camionRepository.findByDisponibleTrue().stream().map(this::toResponse).toList();
    }

    @Transactional(readOnly = true)
    public CamionResponse obtener(Long id) {
        return toResponse(buscar(id));
    }

    public CamionResponse actualizar(Long id, CamionRequest req) {
        Camion c = buscar(id);
        if (!c.getPatente().equals(req.patente()) && camionRepository.existsByPatente(req.patente())) {
            throw new ConflictException("Ya existe un camion con patente " + req.patente());
        }
        if (req.capacidad() != null && req.capacidad() < c.getParadasAsignadas()) {
            throw new BadRequestException("La capacidad no puede ser menor que las paradas ya asignadas ("
                    + c.getParadasAsignadas() + ")");
        }
        aplicar(c, req);
        return toResponse(camionRepository.save(c));
    }

    public CamionResponse cambiarDisponibilidad(Long id, boolean disponible) {
        Camion c = buscar(id);
        c.setDisponible(disponible);
        return toResponse(camionRepository.save(c));
    }

    /** Reserva una parada en el camion. Cuando llega a su capacidad queda no disponible. */
    public CamionResponse asignarParada(Long id) {
        Camion c = buscar(id);
        if (!c.isDisponible()) {
            throw new ConflictException("El camion " + c.getPatente() + " no esta disponible");
        }
        c.setParadasAsignadas(c.getParadasAsignadas() + 1);
        if (c.getParadasAsignadas() >= c.getCapacidad()) {
            c.setDisponible(false);
        }
        return toResponse(camionRepository.save(c));
    }

    /**
     * Compensacion: libera una parada reservada. Se usa si el alta de la parada en ms-rutas
     * falla despues de reservar el cupo, para no dejar la capacidad consumida.
     */
    public CamionResponse liberarParada(Long id) {
        Camion c = buscar(id);
        if (c.getParadasAsignadas() > 0) {
            c.setParadasAsignadas(c.getParadasAsignadas() - 1);
        }
        if (c.getParadasAsignadas() < c.getCapacidad()) {
            c.setDisponible(true);
        }
        return toResponse(camionRepository.save(c));
    }

    public void eliminar(Long id) {
        camionRepository.delete(buscar(id));
    }

    private Camion buscar(Long id) {
        return camionRepository.findById(id)
                .orElseThrow(() -> new NotFoundException("Camion no encontrado: " + id));
    }

    private void aplicar(Camion c, CamionRequest req) {
        c.setPatente(req.patente());
        c.setCapacidad(req.capacidad());
        if (req.zonaId() != null) {
            Zona z = zonaRepository.findById(req.zonaId())
                    .orElseThrow(() -> new NotFoundException("Zona no encontrada: " + req.zonaId()));
            c.setZona(z);
        } else {
            c.setZona(null);
        }
    }

    private CamionResponse toResponse(Camion c) {
        Zona z = c.getZona();
        return new CamionResponse(c.getId(), c.getPatente(), c.getCapacidad(), c.getParadasAsignadas(),
                c.isDisponible(), z == null ? null : z.getId(), z == null ? null : z.getNombre());
    }
}

# ms-rutas - RutaLimpia

Rutas. Hojas de ruta por camion y paradas; el conductor consulta y registra el resultado de cada retiro.

## Requisitos
- Java 17
- Maven 3.9+
- MySQL 8 corriendo en `localhost:3306` (la BD `rutalimpia_rutas` se crea sola)

## Clonar y ejecutar
```bash
git clone <URL_DEL_REPOSITORIO>
cd ms-rutas

# (opcional) credenciales de BD; por defecto root / root
export DB_USER=root
export DB_PASSWORD=root

mvn clean install        # compila, ejecuta tests y empaqueta
mvn spring-boot:run      # levanta el servicio en http://localhost:8084
```

O ejecutando el JAR generado:
```bash
mvn clean package
java -jar target/ms-rutas-1.0.0.jar
```

MySQL rapido con Docker:

## Documentacion
Swagger UI: http://localhost:8084/swagger-ui.html

## Endpoints
| Metodo | Ruta | Descripcion |
|---|---|---|
| POST | /rutas/paradas | Agregar parada a la hoja de ruta (uso interno) |
| GET | /rutas/camion/{camionId}?fecha=YYYY-MM-DD | Hoja de ruta del camion |
| GET | /rutas/paradas/{id} | Obtener parada |
| PUT | /rutas/paradas/{id}/resultado | Registrar REALIZADO / FALLIDO |
| DELETE | /rutas/paradas/{id} | Eliminar parada |

## Estructura
```
controller/  -> API REST
service/     -> logica de negocio
repository/  -> acceso a datos (Spring Data JPA)
model/       -> entidades JPA
dto/         -> objetos de entrada/salida
exception/   -> manejo global de errores
config/      -> configuracion y datos iniciales
```

## Service discovery (Eureka)
Este servicio se registra en Eureka como `ms-rutas`. Levanta primero `eureka-server` (http://localhost:8761).
Para ejecutarlo sin Eureka: `export EUREKA_ENABLED=false`.
En produccion se accede a traves del `api-gateway` (http://localhost:8080).

package com.rutalimpia.rutas.controller;

import java.time.LocalDate;

import org.springframework.format.annotation.DateTimeFormat;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.web.bind.annotation.DeleteMapping;
import org.springframework.web.bind.annotation.GetMapping;
import org.springframework.web.bind.annotation.PathVariable;
import org.springframework.web.bind.annotation.PostMapping;
import org.springframework.web.bind.annotation.PutMapping;
import org.springframework.web.bind.annotation.RequestBody;
import org.springframework.web.bind.annotation.RequestMapping;
import org.springframework.web.bind.annotation.RequestParam;
import org.springframework.web.bind.annotation.RestController;

import com.rutalimpia.rutas.dto.HojaRutaResponse;
import com.rutalimpia.rutas.dto.ParadaRequest;
import com.rutalimpia.rutas.dto.ParadaResponse;
import com.rutalimpia.rutas.dto.ResultadoRequest;
import com.rutalimpia.rutas.service.RutaService;

import jakarta.validation.Valid;
import lombok.RequiredArgsConstructor;

@RestController
@RequestMapping("/rutas")
@RequiredArgsConstructor
public class RutaController {

    private final RutaService rutaService;

    /** Usado por el proceso de asignacion (Lambda). */
    @PostMapping("/paradas")
    public ResponseEntity<ParadaResponse> agregarParada(@Valid @RequestBody ParadaRequest req) {
        return ResponseEntity.status(HttpStatus.CREATED).body(rutaService.agregarParada(req));
    }

    /** Conductor: consulta las paradas asignadas a su camion. */
    @GetMapping("/camion/{camionId}")
    public HojaRutaResponse hojaDelCamion(
            @PathVariable Long camionId,
            @RequestParam(required = false) @DateTimeFormat(iso = DateTimeFormat.ISO.DATE) LocalDate fecha) {
        return rutaService.obtenerHojaPorCamion(camionId, fecha);
    }

    @GetMapping("/paradas/{id}")
    public ParadaResponse obtenerParada(@PathVariable Long id) {
        return rutaService.obtenerParada(id);
    }

    /** Uso interno: parada asociada a una solicitud (para reintentos idempotentes). */
    @GetMapping("/paradas/solicitud/{solicitudId}")
    public ParadaResponse obtenerParadaPorSolicitud(@PathVariable Long solicitudId) {
        return rutaService.obtenerParadaPorSolicitud(solicitudId);
    }

    /** Conductor: registra si el retiro fue realizado o fallo. */
    @PutMapping("/paradas/{id}/resultado")
    public ParadaResponse resultado(@PathVariable Long id, @Valid @RequestBody ResultadoRequest req) {
        return rutaService.registrarResultado(id, req);
    }

    @DeleteMapping("/paradas/{id}")
    public ResponseEntity<Void> eliminar(@PathVariable Long id) {
        rutaService.eliminarParada(id);
        return ResponseEntity.noContent().build();
    }
}

package com.rutalimpia.rutas.exception;

import java.util.stream.Collectors;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.method.annotation.MethodArgumentTypeMismatchException;
import org.springframework.web.servlet.resource.NoResourceFoundException;

@RestControllerAdvice
public class GlobalExceptionHandler {

    private static final Logger log = LoggerFactory.getLogger(GlobalExceptionHandler.class);

    @ExceptionHandler(NotFoundException.class)
    public ResponseEntity<ApiError> notFound(NotFoundException ex) {
        return build(HttpStatus.NOT_FOUND, ex.getMessage());
    }

    @ExceptionHandler(ConflictException.class)
    public ResponseEntity<ApiError> conflict(ConflictException ex) {
        return build(HttpStatus.CONFLICT, ex.getMessage());
    }

    @ExceptionHandler(BadRequestException.class)
    public ResponseEntity<ApiError> badRequest(BadRequestException ex) {
        return build(HttpStatus.BAD_REQUEST, ex.getMessage());
    }

    @ExceptionHandler(UnauthorizedException.class)
    public ResponseEntity<ApiError> unauthorized(UnauthorizedException ex) {
        return build(HttpStatus.UNAUTHORIZED, ex.getMessage());
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiError> validation(MethodArgumentNotValidException ex) {
        String msg = ex.getBindingResult().getFieldErrors().stream()
                .map(f -> f.getField() + ": " + f.getDefaultMessage())
                .collect(Collectors.joining("; "));
        return build(HttpStatus.BAD_REQUEST, msg);
    }

    @ExceptionHandler({HttpMessageNotReadableException.class, MethodArgumentTypeMismatchException.class})
    public ResponseEntity<ApiError> malformed(Exception ex) {
        return build(HttpStatus.BAD_REQUEST, "Solicitud mal formada o parametro invalido");
    }

    @ExceptionHandler(DataIntegrityViolationException.class)
    public ResponseEntity<ApiError> integrity(DataIntegrityViolationException ex) {
        return build(HttpStatus.CONFLICT, "Violacion de integridad de datos (duplicado o referencia invalida)");
    }

    @ExceptionHandler(NoResourceFoundException.class)
    public ResponseEntity<ApiError> noResource(NoResourceFoundException ex) {
        return build(HttpStatus.NOT_FOUND, "Recurso no encontrado");
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<ApiError> generic(Exception ex) {
        log.error("Error interno no controlado", ex);
        return build(HttpStatus.INTERNAL_SERVER_ERROR, "Error interno del servidor");
    }

    private ResponseEntity<ApiError> build(HttpStatus status, String message) {
        return ResponseEntity.status(status).body(ApiError.of(status.value(), status.getReasonPhrase(), message));
    }
}

package com.rutalimpia.rutas.service;

import java.time.LocalDate;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.rutalimpia.rutas.dto.HojaRutaResponse;
import com.rutalimpia.rutas.dto.ParadaRequest;
import com.rutalimpia.rutas.dto.ParadaResponse;
import com.rutalimpia.rutas.dto.ResultadoRequest;
import com.rutalimpia.rutas.exception.BadRequestException;
import com.rutalimpia.rutas.exception.ConflictException;
import com.rutalimpia.rutas.exception.NotFoundException;
import com.rutalimpia.rutas.model.EstadoParada;
import com.rutalimpia.rutas.model.HojaRuta;
import com.rutalimpia.rutas.model.Parada;
import com.rutalimpia.rutas.repository.HojaRutaRepository;
import com.rutalimpia.rutas.repository.ParadaRepository;

import lombok.RequiredArgsConstructor;

@Service
@RequiredArgsConstructor
@Transactional
public class RutaService {

    private final HojaRutaRepository hojaRutaRepository;
    private final ParadaRepository paradaRepository;

    /** Incorpora un retiro a la hoja de ruta del camion (la crea si no existe para esa fecha). */
    public ParadaResponse agregarParada(ParadaRequest req) {
        if (paradaRepository.existsBySolicitudId(req.solicitudId())) {
            throw new ConflictException("La solicitud " + req.solicitudId() + " ya esta en una hoja de ruta");
        }
        LocalDate fecha = req.fecha() != null ? req.fecha() : LocalDate.now();
        HojaRuta hoja = hojaRutaRepository.findByCamionIdAndFecha(req.camionId(), fecha)
                .orElseGet(() -> hojaRutaRepository.save(new HojaRuta(req.camionId(), fecha)));

        Parada p = new Parada();
        p.setSolicitudId(req.solicitudId());
        p.setDireccion(req.direccion());
        p.setTipoResiduo(req.tipoResiduo());
        p.setHorario(req.horario());
        p.setOrden(hoja.getParadas().size() + 1);
        p.setHojaRuta(hoja);
        hoja.getParadas().add(p);
        return toResponse(paradaRepository.save(p));
    }

    /** Hoja de ruta del conductor (por camion). */
    @Transactional(readOnly = true)
    public HojaRutaResponse obtenerHojaPorCamion(Long camionId, LocalDate fecha) {
        LocalDate f = fecha != null ? fecha : LocalDate.now();
        HojaRuta hoja = hojaRutaRepository.findByCamionIdAndFecha(camionId, f)
                .orElseThrow(() -> new NotFoundException(
                        "No hay hoja de ruta para el camion " + camionId + " en la fecha " + f));
        return new HojaRutaResponse(hoja.getId(), hoja.getCamionId(), hoja.getFecha(),
                hoja.getParadas().stream().map(this::toResponse).toList());
    }

    @Transactional(readOnly = true)
    public ParadaResponse obtenerParada(Long id) {
        return toResponse(buscar(id));
    }

    /** Uso interno: permite reintentos idempotentes desde el proceso de asignacion. */
    @Transactional(readOnly = true)
    public ParadaResponse obtenerParadaPorSolicitud(Long solicitudId) {
        return toResponse(paradaRepository.findBySolicitudId(solicitudId)
                .orElseThrow(() -> new NotFoundException("No hay parada para la solicitud " + solicitudId)));
    }

    /** El conductor registra si el retiro fue REALIZADO o FALLIDO. */
    public ParadaResponse registrarResultado(Long id, ResultadoRequest req) {
        if (req.estado() == EstadoParada.PENDIENTE) {
            throw new BadRequestException("El resultado debe ser REALIZADO o FALLIDO");
        }
        Parada p = buscar(id);
        p.setEstado(req.estado());
        p.setObservacion(req.observacion());
        return toResponse(paradaRepository.save(p));
    }

    public void eliminarParada(Long id) {
        Parada p = buscar(id);
        p.getHojaRuta().getParadas().remove(p);
        paradaRepository.delete(p);
    }

    private Parada buscar(Long id) {
        return paradaRepository.findById(id)
                .orElseThrow(() -> new NotFoundException("Parada no encontrada: " + id));
    }

    private ParadaResponse toResponse(Parada p) {
        return new ParadaResponse(p.getId(), p.getSolicitudId(), p.getHojaRuta().getCamionId(),
                p.getDireccion(), p.getTipoResiduo(), p.getHorario(), p.getOrden(), p.getEstado(), p.getObservacion());
    }
}

# ms-solicitudes - RutaLimpia

Solicitudes de retiro. Registra la solicitud al instante (folio) y la asignacion de camion se procesa de forma asincrona (worker que simula SQS + Lambda). Requiere ms-flota (8083) y ms-rutas (8084) para asignar.

## Requisitos
- Java 17
- Maven 3.9+
- MySQL 8 corriendo en `localhost:3306` (la BD `rutalimpia_solicitudes` se crea sola)

## Clonar y ejecutar
```bash
git clone <URL_DEL_REPOSITORIO>
cd ms-solicitudes

# (opcional) credenciales de BD; por defecto root / root
export DB_USER=root
export DB_PASSWORD=root

mvn clean install        # compila, ejecuta tests y empaqueta
mvn spring-boot:run      # levanta el servicio en http://localhost:8082
```

O ejecutando el JAR generado:
```bash
mvn clean package
java -jar target/ms-solicitudes-1.0.0.jar
```

MySQL rapido con Docker:
```bash
docker run -d --name mysql-rl -e MYSQL_ROOT_PASSWORD=root -p 3306:3306 mysql:8
```

## Documentacion
Swagger UI: http://localhost:8082/swagger-ui.html

## Endpoints
| Metodo | Ruta | Descripcion |
|---|---|---|
| POST | /solicitudes | Crear solicitud (devuelve folio, estado PENDIENTE) |
| GET | /solicitudes?usuarioId= | Listar solicitudes (del vecino) |
| GET | /solicitudes/folio/{folio} | Consultar por folio |
| PUT | /solicitudes/{id} | Modificar (solo PENDIENTE) |
| PUT | /solicitudes/{id}/estado | Cambiar estado |
| DELETE | /solicitudes/{id} | Cancelar (solo PENDIENTE) |
| GET/POST | /tipos-residuo | Listar / crear tipos de residuo |

## Estructura
```
controller/  -> API REST
service/     -> logica de negocio
repository/  -> acceso a datos (Spring Data JPA)
model/       -> entidades JPA
dto/         -> objetos de entrada/salida
exception/   -> manejo global de errores
config/      -> configuracion y datos iniciales
```

## Service discovery (Eureka)
Este servicio se registra en Eureka como `ms-solicitudes`. Levanta primero `eureka-server` (http://localhost:8761).
Para ejecutarlo sin Eureka: `export EUREKA_ENABLED=false`.
En produccion se accede a traves del `api-gateway` (http://localhost:8080).

package com.rutalimpia.solicitudes.client;

import java.util.List;
import java.util.Optional;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.core.ParameterizedTypeReference;
import org.springframework.http.client.SimpleClientHttpRequestFactory;
import org.springframework.stereotype.Component;
import org.springframework.web.client.RestClient;

/** Cliente HTTP hacia ms-flota (resuelto por Eureka). */
@Component
public class FlotaClient {

    private final RestClient restClient;

    public FlotaClient(RestClient.Builder builder, @Value("${servicios.flota.url}") String baseUrl) {
        SimpleClientHttpRequestFactory factory = new SimpleClientHttpRequestFactory();
        factory.setConnectTimeout(2000);
        factory.setReadTimeout(3000);
        this.restClient = builder.clone().baseUrl(baseUrl).requestFactory(factory).build();
    }

    public Optional<CamionDto> primerCamionDisponible() {
        List<CamionDto> lista = restClient.get().uri("/camiones/disponibles").retrieve()
                .body(new ParameterizedTypeReference<List<CamionDto>>() { });
        return lista == null || lista.isEmpty() ? Optional.empty() : Optional.of(lista.get(0));
    }

    public void reservarParada(Long camionId) {
        restClient.put().uri("/camiones/{id}/asignar-parada", camionId).retrieve().toBodilessEntity();
    }

    /** Compensacion: devuelve el cupo reservado si la parada no se pudo crear. */
    public void liberarParada(Long camionId) {
        restClient.put().uri("/camiones/{id}/liberar-parada", camionId).retrieve().toBodilessEntity();
    }
}

package com.rutalimpia.solicitudes.client;

import java.util.Optional;

import org.springframework.beans.factory.annotation.Value;
import org.springframework.http.client.SimpleClientHttpRequestFactory;
import org.springframework.stereotype.Component;
import org.springframework.web.client.HttpClientErrorException;
import org.springframework.web.client.RestClient;

/** Cliente HTTP hacia ms-rutas (resuelto por Eureka). */
@Component
public class RutasClient {

    private final RestClient restClient;

    public RutasClient(RestClient.Builder builder, @Value("${servicios.rutas.url}") String baseUrl) {
        SimpleClientHttpRequestFactory factory = new SimpleClientHttpRequestFactory();
        factory.setConnectTimeout(2000);
        factory.setReadTimeout(3000);
        this.restClient = builder.clone().baseUrl(baseUrl).requestFactory(factory).build();
    }

    public void agregarParada(ParadaRequestDto dto) {
        restClient.post().uri("/rutas/paradas").body(dto).retrieve().toBodilessEntity();
    }

    /** Devuelve la parada de una solicitud si ya existe (para reintentos idempotentes). */
    public Optional<ParadaDto> buscarPorSolicitud(Long solicitudId) {
        try {
            ParadaDto dto = restClient.get().uri("/rutas/paradas/solicitud/{id}", solicitudId)
                    .retrieve().body(ParadaDto.class);
            return Optional.ofNullable(dto);
        } catch (HttpClientErrorException.NotFound e) {
            return Optional.empty();
        }
    }
}

package com.rutalimpia.solicitudes.exception;

import java.util.stream.Collectors;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.dao.DataIntegrityViolationException;
import org.springframework.http.HttpStatus;
import org.springframework.http.ResponseEntity;
import org.springframework.http.converter.HttpMessageNotReadableException;
import org.springframework.web.bind.MethodArgumentNotValidException;
import org.springframework.web.bind.annotation.ExceptionHandler;
import org.springframework.web.bind.annotation.RestControllerAdvice;
import org.springframework.web.method.annotation.MethodArgumentTypeMismatchException;
import org.springframework.web.servlet.resource.NoResourceFoundException;

@RestControllerAdvice
public class GlobalExceptionHandler {

    private static final Logger log = LoggerFactory.getLogger(GlobalExceptionHandler.class);

    @ExceptionHandler(NotFoundException.class)
    public ResponseEntity<ApiError> notFound(NotFoundException ex) {
        return build(HttpStatus.NOT_FOUND, ex.getMessage());
    }

    @ExceptionHandler(ConflictException.class)
    public ResponseEntity<ApiError> conflict(ConflictException ex) {
        return build(HttpStatus.CONFLICT, ex.getMessage());
    }

    @ExceptionHandler(BadRequestException.class)
    public ResponseEntity<ApiError> badRequest(BadRequestException ex) {
        return build(HttpStatus.BAD_REQUEST, ex.getMessage());
    }

    @ExceptionHandler(UnauthorizedException.class)
    public ResponseEntity<ApiError> unauthorized(UnauthorizedException ex) {
        return build(HttpStatus.UNAUTHORIZED, ex.getMessage());
    }

    @ExceptionHandler(MethodArgumentNotValidException.class)
    public ResponseEntity<ApiError> validation(MethodArgumentNotValidException ex) {
        String msg = ex.getBindingResult().getFieldErrors().stream()
                .map(f -> f.getField() + ": " + f.getDefaultMessage())
                .collect(Collectors.joining("; "));
        return build(HttpStatus.BAD_REQUEST, msg);
    }

    @ExceptionHandler({HttpMessageNotReadableException.class, MethodArgumentTypeMismatchException.class})
    public ResponseEntity<ApiError> malformed(Exception ex) {
        return build(HttpStatus.BAD_REQUEST, "Solicitud mal formada o parametro invalido");
    }

    @ExceptionHandler(DataIntegrityViolationException.class)
    public ResponseEntity<ApiError> integrity(DataIntegrityViolationException ex) {
        return build(HttpStatus.CONFLICT, "Violacion de integridad de datos (duplicado o referencia invalida)");
    }

    @ExceptionHandler(NoResourceFoundException.class)
    public ResponseEntity<ApiError> noResource(NoResourceFoundException ex) {
        return build(HttpStatus.NOT_FOUND, "Recurso no encontrado");
    }

    @ExceptionHandler(Exception.class)
    public ResponseEntity<ApiError> generic(Exception ex) {
        log.error("Error interno no controlado", ex);
        return build(HttpStatus.INTERNAL_SERVER_ERROR, "Error interno del servidor");
    }

    private ResponseEntity<ApiError> build(HttpStatus status, String message) {
        return ResponseEntity.status(status).body(ApiError.of(status.value(), status.getReasonPhrase(), message));
    }
}

    /** Referencia logica a ms-flota; se completa cuando se asigna. */
    private Long camionId;

    @Column(nullable = false)
    private LocalDateTime fechaCreacion = LocalDateTime.now();

    @ManyToOne(fetch = FetchType.LAZY, optional = false)
    @JoinColumn(name = "tipo_residuo_id", nullable = false)

package com.rutalimpia.solicitudes.service;

import java.util.List;
import java.util.Optional;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.scheduling.annotation.Scheduled;
import org.springframework.stereotype.Component;

import com.rutalimpia.solicitudes.client.CamionDto;
import com.rutalimpia.solicitudes.client.FlotaClient;
import com.rutalimpia.solicitudes.client.ParadaDto;
import com.rutalimpia.solicitudes.client.ParadaRequestDto;
import com.rutalimpia.solicitudes.client.RutasClient;
import com.rutalimpia.solicitudes.model.EstadoSolicitud;
import com.rutalimpia.solicitudes.model.SolicitudRetiro;
import com.rutalimpia.solicitudes.repository.SolicitudRepository;

import lombok.RequiredArgsConstructor;

/**
 * Simula la funcion AWS Lambda que consume la cola SQS:
 *  1. toma solicitudes PENDIENTES
 *  2. consulta disponibilidad a ms-flota
 *  3. agrega la parada a la hoja de ruta en ms-rutas
 *  4. marca la solicitud como ASIGNADA
 * Si no hay camion o un servicio no responde, la solicitud sigue PENDIENTE y se reintenta en la siguiente corrida.
 */
@Component
@RequiredArgsConstructor
@ConditionalOnProperty(name = "rutalimpia.worker.enabled", havingValue = "true", matchIfMissing = true)
public class AsignacionWorker {

    private static final Logger log = LoggerFactory.getLogger(AsignacionWorker.class);

    private final SolicitudRepository solicitudRepository;
    private final SolicitudService solicitudService;
    private final FlotaClient flotaClient;
    private final RutasClient rutasClient;

    @Scheduled(fixedDelayString = "${rutalimpia.worker.delay-ms:10000}")
    public void procesarPendientes() {
        List<SolicitudRetiro> pendientes =
                solicitudRepository.findByEstadoOrderByFechaCreacionAsc(EstadoSolicitud.PENDIENTE);
        if (pendientes.isEmpty()) {
            return;
        }
        log.info("[WORKER] {} solicitud(es) pendiente(s)", pendientes.size());

        for (SolicitudRetiro s : pendientes) {
            try {
                if (procesar(s)) {
                    break; // sin camiones disponibles: no tiene sentido seguir con el resto
                }
            } catch (Exception e) {
                // Un fallo en una solicitud no debe abortar las demas.
                log.warn("[WORKER] No se pudo asignar {} (se reintentara). Detalle: {}", s.getFolio(), e.getMessage());
            }
        }
    }

    /** @return true si no hay camiones disponibles (conviene detener el lote). */
    private boolean procesar(SolicitudRetiro s) {
        // Idempotencia: si la parada ya existe (p.ej. reintento tras un timeout) solo marcamos la solicitud.
        Optional<ParadaDto> existente = rutasClient.buscarPorSolicitud(s.getId());
        if (existente.isPresent() && existente.get().camionId() != null) {
            solicitudService.marcarAsignada(s.getId(), existente.get().camionId());
            log.info("[WORKER] {} ya tenia parada creada; marcada como ASIGNADA", s.getFolio());
            return false;
        }

        Optional<CamionDto> camion = flotaClient.primerCamionDisponible();
        if (camion.isEmpty()) {
            log.warn("[WORKER] Sin camiones disponibles. {} sigue PENDIENTE", s.getFolio());
            return true;
        }

        Long camionId = camion.get().id();
        flotaClient.reservarParada(camionId);
        try {
            rutasClient.agregarParada(new ParadaRequestDto(s.getId(), camionId, s.getDireccion(),
                    s.getTipoResiduo().getNombre(), s.getHorarioPreferido()));
        } catch (RuntimeException e) {
            // Compensacion: si no se creo la parada, devolvemos el cupo reservado en ms-flota.
            flotaClient.liberarParada(camionId);
            throw e;
        }
        solicitudService.marcarAsignada(s.getId(), camionId);
        log.info("[WORKER] {} asignada al camion {}", s.getFolio(), camion.get().patente());
        return false;
    }
}

package com.rutalimpia.solicitudes.service;

import java.time.LocalDate;
import java.time.format.DateTimeFormatter;
import java.util.List;
import java.util.UUID;

import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import com.rutalimpia.solicitudes.dto.EstadoRequest;
import com.rutalimpia.solicitudes.dto.SolicitudRequest;
import com.rutalimpia.solicitudes.dto.SolicitudResponse;
import com.rutalimpia.solicitudes.dto.SolicitudUpdateRequest;
import com.rutalimpia.solicitudes.exception.ConflictException;
import com.rutalimpia.solicitudes.exception.NotFoundException;
import com.rutalimpia.solicitudes.messaging.AsignacionPublisher;
import com.rutalimpia.solicitudes.model.EstadoSolicitud;
import com.rutalimpia.solicitudes.model.SolicitudRetiro;
import com.rutalimpia.solicitudes.model.TipoResiduo;
import com.rutalimpia.solicitudes.repository.SolicitudRepository;
import com.rutalimpia.solicitudes.repository.TipoResiduoRepository;

import lombok.RequiredArgsConstructor;

@Service
@RequiredArgsConstructor
@Transactional
public class SolicitudService {

    private final SolicitudRepository solicitudRepository;
    private final TipoResiduoRepository tipoResiduoRepository;
    private final AsignacionPublisher asignacionPublisher;

    /**
     * Registra la solicitud de inmediato (estado PENDIENTE) y devuelve el folio de recepcion.
     * La asignacion del camion ocurre despues, de forma asincrona, sin bloquear al vecino.
     */
    public SolicitudResponse crear(SolicitudRequest req) {
        TipoResiduo tipo = tipoResiduoRepository.findById(req.tipoResiduoId())
                .orElseThrow(() -> new NotFoundException("Tipo de residuo no encontrado: " + req.tipoResiduoId()));

        SolicitudRetiro s = new SolicitudRetiro();
        s.setFolio(generarFolio());
        s.setUsuarioId(req.usuarioId());
        s.setDireccion(req.direccion());
        s.setHorarioPreferido(req.horarioPreferido());
        s.setTipoResiduo(tipo);
        s.setEstado(EstadoSolicitud.PENDIENTE);
        s = solicitudRepository.save(s);

        asignacionPublisher.publicar(s.getFolio());
        return toResponse(s);
    }

    @Transactional(readOnly = true)
    public SolicitudResponse obtenerPorFolio(String folio) {
        return toResponse(solicitudRepository.findByFolio(folio)
                .orElseThrow(() -> new NotFoundException("Folio no encontrado: " + folio)));
    }

    @Transactional(readOnly = true)
    public List<SolicitudResponse> listar(Long usuarioId) {
        List<SolicitudRetiro> lista = usuarioId == null
                ? solicitudRepository.findAll()
                : solicitudRepository.findByUsuarioIdOrderByFechaCreacionDesc(usuarioId);
        return lista.stream().map(this::toResponse).toList();
    }

    public SolicitudResponse actualizar(Long id, SolicitudUpdateRequest req) {
        SolicitudRetiro s = buscar(id);
        if (s.getEstado() != EstadoSolicitud.PENDIENTE) {
            throw new ConflictException("Solo se pueden modificar solicitudes PENDIENTES");
        }
        s.setDireccion(req.direccion());
        s.setHorarioPreferido(req.horarioPreferido());
        return toResponse(solicitudRepository.save(s));
    }

    public SolicitudResponse cambiarEstado(Long id, EstadoRequest req) {
        SolicitudRetiro s = buscar(id);
        s.setEstado(req.estado());
        if (req.camionId() != null) {
            s.setCamionId(req.camionId());
        }
        return toResponse(solicitudRepository.save(s));
    }

    /** Usado por el worker de asignacion. */
    public void marcarAsignada(Long id, Long camionId) {
        SolicitudRetiro s = buscar(id);
        s.setEstado(EstadoSolicitud.ASIGNADA);
        s.setCamionId(camionId);
        solicitudRepository.save(s);
    }

    public void cancelar(Long id) {
        SolicitudRetiro s = buscar(id);
        if (s.getEstado() != EstadoSolicitud.PENDIENTE) {
            throw new ConflictException("Solo se pueden cancelar solicitudes PENDIENTES");
        }
        solicitudRepository.delete(s);
    }

    private SolicitudRetiro buscar(Long id) {
        return solicitudRepository.findById(id)
                .orElseThrow(() -> new NotFoundException("Solicitud no encontrada: " + id));
    }

    private String generarFolio() {
        String fecha = LocalDate.now().format(DateTimeFormatter.BASIC_ISO_DATE);
        return "RL-" + fecha + "-" + UUID.randomUUID().toString().substring(0, 6).toUpperCase();
    }

    private SolicitudResponse toResponse(SolicitudRetiro s) {
        return new SolicitudResponse(s.getId(), s.getFolio(), s.getUsuarioId(), s.getDireccion(),
                s.getTipoResiduo().getNombre(), s.getHorarioPreferido(), s.getEstado(),
                s.getCamionId(), s.getFechaCreacion());
    }
}

package com.rutalimpia.solicitudes.service;

import java.util.List;

import org.springframework.stereotype.Service;

import com.rutalimpia.solicitudes.dto.TipoResiduoRequest;
import com.rutalimpia.solicitudes.exception.ConflictException;
import com.rutalimpia.solicitudes.model.TipoResiduo;
import com.rutalimpia.solicitudes.repository.TipoResiduoRepository;

import lombok.RequiredArgsConstructor;

@Service
@RequiredArgsConstructor
public class TipoResiduoService {

    private final TipoResiduoRepository repository;

    public List<TipoResiduo> listar() {
        return repository.findAll();
    }

    public TipoResiduo crear(TipoResiduoRequest req) {
        if (repository.existsByNombre(req.nombre())) {
            throw new ConflictException("Ya existe el tipo de residuo " + req.nombre());
        }
        return repository.save(new TipoResiduo(req.nombre()));
    }
}
